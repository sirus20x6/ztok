   TOKEN=$(kubectl get secret {{ include "ztok.authSecretName" . }} -n {{ .Release.Namespace }} -o jsonpath='{.data.token}' | base64 -d)

        http://{{ include "ztok.fullname" . }}.{{ .Release.Namespace }}.svc.cluster.local:{{ .Values.service.port }}/encode \
        -d '{"text":"hello world"}'
